===============================================================================================
ЗАМЕНА 1 (парковка). Файл: Assets/_Project/01_Scripts/Editor/KvXrTools.cs
===============================================================================================

--- БЫЛО (внутри try, сразу после открывающей скобки try) -----------------------------------

            if (Directory.Exists(testsPath))
            {
                try
                {
                    if (Directory.Exists(testsParked)) Directory.Delete(testsParked, true);
                    Directory.CreateDirectory(Path.GetDirectoryName(testsParked));
                    Directory.Move(testsPath, testsParked);
                    testsMoved = true;
                    Debug.Log("[KvXr] Тесты временно убраны из Assets на время сборки APK (в сборке плеера их нет).");
                }
                catch (Exception e)
                {
                    Debug.LogWarning("[KvXr] Не удалось убрать тесты из сборки: " + e.Message);
                }
            }

--- СТАЛО ------------------------------------------------------------------------------------

            try
            {
                TestParkingReport parked = ParkTestsForBuild(testsPath, testsParked);
                testsMoved = !parked.NothingToDo;

                if (parked.Success)
                    Debug.Log("[KvXr] Тесты убраны из сборки APK: " + parked.Describe());
                else
                    Debug.LogWarning("[KvXr] Тесты НЕ убраны из сборки APK: " + parked.Describe());
            }
            catch (Exception e)
            {
                Debug.LogWarning("[KvXr] Не удалось убрать тесты из сборки: " + e.Message);
            }

ПОЯСНЕНИЕ: `testsMoved` теперь означает «парковка состоялась, возврат нужен». Если папки
тестов в Assets не было вовсе, `NothingToDo` = true и возвращать нечего — как и раньше.
